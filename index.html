var SPREADSHEET_ID = '1XCmfA9swRSGV7RmzGpDq3Bz0YesYGPtRrRMWhYkKzIE';

function getTargetSpreadsheet() {
  try {
    var active = SpreadsheetApp.getActiveSpreadsheet();
    if (active) return active;
  } catch (e) {}
  return SpreadsheetApp.openById(SPREADSHEET_ID);
}

function doGet(e) {
  var isApi = e && e.parameter && e.parameter.mode === 'api';
  if (!isApi) {
    try {
      return HtmlService.createHtmlOutputFromFile('Index')
        .setTitle('ระบบจองห้องเรียน (Smart School Room Booking)')
        .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    } catch (err1) {
      try {
        return HtmlService.createHtmlOutputFromFile('index')
          .setTitle('ระบบจองห้องเรียน (Smart School Room Booking)')
          .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
      } catch (err2) {
        // หากยังไม่ได้สร้างไฟล์ Index.html จะส่งข้อมูล JSON API แทนโดยไม่เกิด Error
      }
    }
  }
  return ContentService
    .createTextOutput(JSON.stringify(getAppData()))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  if (!e || !e.postData || !e.postData.contents) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ready', message: 'doPost พร้อมทำงาน' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var payload = JSON.parse(e.postData.contents);
  saveAppData(payload);
  return ContentService
    .createTextOutput(JSON.stringify({ status: 'saved' }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getAppData() {
  var ss = getTargetSpreadsheet();
  return {
    status: 'ok',
    rooms: readSheetData(ss, 'Rooms'),
    bookings: readSheetData(ss, 'Bookings'),
    users: readSheetData(ss, 'Users')
  };
}

function saveAppData(payload) {
  if (!payload) return { status: 'no_payload' };
  var ss = getTargetSpreadsheet();
  if (payload.rooms) {
    writeSheetData(ss, 'Rooms', ['roomId','name','building','capacity','category','equipment','requireApproval','status'], payload.rooms);
  }
  if (payload.bookings) {
    writeSheetData(ss, 'Bookings', ['id','roomId','date','startPeriod','endPeriod','teacher','department','title','classGroup','equipment','note','status','createdAt'], payload.bookings);
  }
  if (payload.users) {
    writeSheetData(ss, 'Users', ['username','password','fullName','department','role'], payload.users);
  }
  return { status: 'saved' };
}

function readSheetData(ss, sheetName) {
  if (!ss) {
    ss = getTargetSpreadsheet();
  }
  if (!sheetName) {
    sheetName = 'Rooms';
  }
  var sh = ss.getSheetByName(sheetName);
  if (!sh) return [];
  var rows = sh.getDataRange().getValues();
  if (rows.length === 0) return [];
  var headers = rows.shift();
  var result = [];
  for (var r = 0; r !== rows.length; r++) {
    var obj = {};
    for (var c = 0; c !== headers.length; c++) {
      obj[headers[c]] = rows[r][c];
    }
    result.push(obj);
  }
  return result;
}

function writeSheetData(ss, sheetName, headers, items) {
  if (!ss) {
    ss = getTargetSpreadsheet();
  }
  if (!sheetName || !headers || !items) return;
  var sh = ss.getSheetByName(sheetName);
  if (!sh) {
    sh = ss.insertSheet(sheetName);
  }
  sh.clearContents();
  sh.appendRow(headers);
  for (var i = 0; i !== items.length; i++) {
    var item = items[i];
    var row = [];
    for (var j = 0; j !== headers.length; j++) {
      var key = headers[j];
      var val = item[key];
      if (Array.isArray(val)) {
        row.push(val.join(', '));
      } else if (val === undefined || val === null) {
        row.push('');
      } else {
        row.push(val);
      }
    }
    sh.appendRow(row);
  }
}
